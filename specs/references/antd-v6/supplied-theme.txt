/**
 * Get theme for specific density
 */
export const getThemeByDensity = (density: 'default' | 'compact'): ThemeConfig => {
    if (density === 'compact') {
        return {
            ...cssVarTheme,
            token: {
                ...cssVarTheme.token,
                // @ts-ignore FIX ME
                controlHeight: cssVar('--control-height-default'), // 28 in compact
                // @ts-ignore FIX ME
                controlHeightSM: cssVar('--control-height-sm'), // 21 in compact
                // @ts-ignore FIX ME
                controlHeightLG: cssVar('--control-height-lg'), // 35 in compact
                // @ts-ignore FIX ME
                controlHeightXS: cssVar('--control-height-xs'), // 14 in compact
                controlInteractiveSize: 14,
            },
        };
    }
    return cssVarTheme;
};
